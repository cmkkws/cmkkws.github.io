23:T2cf3,"""Attention Is All You Need: original encoder-decoder architecture.

Save this file as transformer_2017.py. Run: python transformer_2017.py --smoke
Default dimensions are the paper's base configuration. Smoke inputs use a tiny
configuration and synthetic token sequences, not the WMT translation benchmark.
"""
import argparse
import math
from dataclasses import dataclass

import torch
from torch import nn
from torch.nn import functional as F


# STEP 01: configuration and teacher forcing
@dataclass
class Config:
    vocab_size: int
    d_model: int = 512
    heads: int = 8
    layers: int = 6
    d_ff: int = 2048
    dropout: float = 0.1
    max_length: int = 512
    pad_id: int = 0
    bos_id: int = 1
    eos_id: int = 2

    def __post_init__(self):
        if self.vocab_size < 4 or self.d_model < 2 or self.d_model % 2:
            raise ValueError("vocab_size >= 4 and positive even d_model required")
        if self.heads < 1 or self.d_model % self.heads:
            raise ValueError("d_model must be divisible by heads")
        if self.layers < 1 or self.d_ff < 1 or self.max_length < 1:
            raise ValueError("layers, d_ff and max_length must be positive")
        if not 0 <= self.dropout < 1:
            raise ValueError("dropout must be in [0, 1)")
        if len({self.pad_id, self.bos_id, self.eos_id}) != 3:
            raise ValueError("PAD, BOS, EOS must be distinct")
        if any(not 0 <= i < self.vocab_size for i in (self.pad_id, self.bos_id, self.eos_id)):
            raise ValueError("special IDs must be inside vocabulary")


def teacher_forcing(target):
    if target.ndim != 2 or target.size(1) < 2:
        raise ValueError("target must be [batch, length >= 2]")
    return target[:, :-1], target[:, 1:]  # shift exactly once


# STEP 02: scaled embeddings and sinusoidal positions
class TokenPosition(nn.Module):
    def __init__(self, cfg, embedding):
        super().__init__()
        self.embedding = embedding
        self.scale = math.sqrt(cfg.d_model)
        position = torch.arange(cfg.max_length, dtype=torch.float32)[:, None]
        frequency = torch.exp(
            torch.arange(0, cfg.d_model, 2) * (-math.log(10000.0) / cfg.d_model)
        )
        pe = torch.zeros(cfg.max_length, cfg.d_model)
        pe[:, 0::2] = torch.sin(position * frequency)
        pe[:, 1::2] = torch.cos(position * frequency)
        self.register_buffer("pe", pe[None])
        self.dropout = nn.Dropout(cfg.dropout)

    def forward(self, ids):
        if ids.size(1) > self.pe.size(1):
            raise ValueError("sequence exceeds positional encoding length")
        x = self.embedding(ids) * self.scale
        return self.dropout(x + self.pe[:, :ids.size(1)].to(x.dtype))


# STEP 03: padding masks, causal masks, scaled dot-product attention
def key_mask(ids, pad_id):
    return (ids != pad_id)[:, None, None, :]  # True = allowed


def target_mask(ids, pad_id):
    length = ids.size(1)
    causal = torch.ones(length, length, dtype=torch.bool, device=ids.device).tril()
    return key_mask(ids, pad_id) & causal[None, None]


def scaled_attention(q, k, v, allowed):
    scores = q @ k.transpose(-2, -1) / math.sqrt(q.size(-1))
    has_key = allowed.any(dim=-1, keepdim=True)
    scores = scores.masked_fill(~allowed, float("-inf"))
    # An empty allowed row is an explicit zero-output policy, not a NaN softmax.
    scores = torch.where(has_key, scores, torch.zeros_like(scores))
    probabilities = torch.softmax(scores, dim=-1).masked_fill(~allowed, 0.0)
    return probabilities @ v


# STEP 04: multi-head self-attention and cross-attention
class MultiHeadAttention(nn.Module):
    def __init__(self, cfg):
        super().__init__()
        self.heads = cfg.heads
        self.head_dim = cfg.d_model // cfg.heads
        self.q = nn.Linear(cfg.d_model, cfg.d_model, bias=False)
        self.k = nn.Linear(cfg.d_model, cfg.d_model, bias=False)
        self.v = nn.Linear(cfg.d_model, cfg.d_model, bias=False)
        self.out = nn.Linear(cfg.d_model, cfg.d_model, bias=False)

    def split(self, x):
        batch, length, _ = x.shape
        return x.reshape(batch, length, self.heads, self.head_dim).transpose(1, 2)

    def forward(self, query, key, value, allowed):
        q, k, v = self.split(self.q(query)), self.split(self.k(key)), self.split(self.v(value))
        context = scaled_attention(q, k, v, allowed)
        merged = context.transpose(1, 2).contiguous().reshape(query.size(0), query.size(1), -1)
        return self.out(merged)


# STEP 05: ReLU FFN and original post-normalization encoder
class FeedForward(nn.Module):
    def __init__(self, cfg):
        super().__init__()
        self.first = nn.Linear(cfg.d_model, cfg.d_ff)
        self.second = nn.Linear(cfg.d_ff, cfg.d_model)

    def forward(self, x):
        return self.second(F.relu(self.first(x)))


class EncoderLayer(nn.Module):
    def __init__(self, cfg):
        super().__init__()
        self.attention = MultiHeadAttention(cfg)
        self.ffn = FeedForward(cfg)
        self.norm1 = nn.LayerNorm(cfg.d_model, eps=1e-5)
        self.norm2 = nn.LayerNorm(cfg.d_model, eps=1e-5)
        self.dropout = nn.Dropout(cfg.dropout)

    def forward(self, x, allowed):
        x = self.norm1(x + self.dropout(self.attention(x, x, x, allowed)))
        return self.norm2(x + self.dropout(self.ffn(x)))


# STEP 06: original decoder with encoder-decoder attention
class DecoderLayer(nn.Module):
    def __init__(self, cfg):
        super().__init__()
        self.self_attention = MultiHeadAttention(cfg)
        self.cross_attention = MultiHeadAttention(cfg)
        self.ffn = FeedForward(cfg)
        self.norm1 = nn.LayerNorm(cfg.d_model, eps=1e-5)
        self.norm2 = nn.LayerNorm(cfg.d_model, eps=1e-5)
        self.norm3 = nn.LayerNorm(cfg.d_model, eps=1e-5)
        self.dropout = nn.Dropout(cfg.dropout)

    def forward(self, x, memory, causal_allowed, source_allowed):
        x = self.norm1(x + self.dropout(self.self_attention(x, x, x, causal_allowed)))
        x = self.norm2(x + self.dropout(self.cross_attention(x, memory, memory, source_allowed)))
        return self.norm3(x + self.dropout(self.ffn(x)))


# STEP 07: complete encoder-decoder and shared output weights
class Transformer2017(nn.Module):
    def __init__(self, cfg):
        super().__init__()
        self.cfg = cfg
        # One joint source/target vocabulary is assumed for three-way weight tying.
        embedding = nn.Embedding(cfg.vocab_size, cfg.d_model, padding_idx=cfg.pad_id)
        self.source_position = TokenPosition(cfg, embedding)
        self.target_position = TokenPosition(cfg, embedding)
        self.encoder = nn.ModuleList([EncoderLayer(cfg) for _ in range(cfg.layers)])
        self.decoder = nn.ModuleList([DecoderLayer(cfg) for _ in range(cfg.layers)])
        self.generator = nn.Linear(cfg.d_model, cfg.vocab_size, bias=False)
        self.generator.weight = embedding.weight
        # Xavier and LayerNorm epsilon are documented implementation choices.
        for parameter in self.parameters():
            if parameter.dim() > 1:
                nn.init.xavier_uniform_(parameter)
        with torch.no_grad():
            embedding.weight[cfg.pad_id].zero_()

    def encode(self, source):
        allowed = key_mask(source, self.cfg.pad_id)
        x = self.source_position(source)
        for layer in self.encoder:
            x = layer(x, allowed)
        return x

    def decode(self, target_input, memory, source):
        causal_allowed = target_mask(target_input, self.cfg.pad_id)
        source_allowed = key_mask(source, self.cfg.pad_id)
        x = self.target_position(target_input)
        for layer in self.decoder:
            x = layer(x, memory, causal_allowed, source_allowed)
        return self.generator(x)  # logits; do not softmax before training loss

    def forward(self, source, target_input):
        return self.decode(target_input, self.encode(source), source)


# STEP 08: label smoothing and the paper's learning-rate schedule
def label_smoothed_loss(logits, labels, pad_id=0, smoothing=0.1):
    if not 0 <= smoothing < 1:
        raise ValueError("smoothing must be in [0, 1)")
    valid = labels != pad_id
    if not valid.any():
        return logits.sum() * 0.0
    log_probs = F.log_softmax(logits[valid], dim=-1)
    targets = labels[valid]
    nll = -log_probs.gather(1, targets[:, None]).squeeze(1)
    # Confidence=1-epsilon; epsilon distributed over all non-target vocabulary IDs.
    other = (-log_probs.sum(dim=-1) - nll) / (logits.size(-1) - 1)
    return ((1 - smoothing) * nll + smoothing * other).mean()


def noam_lr(step, d_model, warmup=4000):
    if step < 1 or d_model < 1 or warmup < 1:
        raise ValueError("step, d_model and warmup must be positive")
    return d_model ** -0.5 * min(step ** -0.5, step * warmup ** -1.5)


# STEP 09: one training update on paired sequences
def train_step(model, optimizer, source, target, step, warmup=4000):
    model.train()
    target_input, labels = teacher_forcing(target)
    optimizer.zero_grad(set_to_none=True)
    logits = model(source, target_input)
    loss = label_smoothed_loss(logits, labels, model.cfg.pad_id)
    if not torch.isfinite(loss):
        raise RuntimeError("non-finite loss")
    loss.backward()
    for group in optimizer.param_groups:
        group["lr"] = noam_lr(step, model.cfg.d_model, warmup)
    optimizer.step()
    return float(loss.detach())


def toy_batch():
    source = torch.tensor([[3, 4, 5, 0], [6, 7, 0, 0]])
    target = torch.tensor([[1, 5, 4, 3, 2], [1, 7, 6, 2, 0]])
    return source, target


# STEP 10: greedy autoregressive decoding and smoke command
@torch.no_grad()
def greedy_decode(model, source, max_new_tokens=8):
    if max_new_tokens < 1 or max_new_tokens > model.cfg.max_length:
        raise ValueError("max_new_tokens outside positional capacity")
    was_training = model.training
    model.eval()
    try:
        memory = model.encode(source)
        ids = torch.full((source.size(0), 1), model.cfg.bos_id, dtype=torch.long, device=source.device)
        finished = torch.zeros(source.size(0), dtype=torch.bool, device=source.device)
        for _ in range(max_new_tokens):
            token = model.decode(ids, memory, source)[:, -1].argmax(dim=-1)
            token = torch.where(finished, model.cfg.pad_id, token)
            ids = torch.cat([ids, token[:, None]], dim=1)
            finished |= token == model.cfg.eos_id
            if finished.all():
                break
        return ids
    finally:
        model.train(was_training)


def smoke():

    torch.manual_seed(42)

    print("greedy IDs:", greedy_decode(model, source).tolist())
